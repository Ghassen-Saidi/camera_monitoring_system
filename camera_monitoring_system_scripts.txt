package camera;

import javafx.application.Platform;
import javafx.collections.FXCollections;
import javafx.collections.ObservableList;
import javafx.concurrent.Task;
import javafx.fxml.FXML;
import javafx.scene.control.*;
import javafx.scene.control.cell.PropertyValueFactory;

import java.util.List;

public class CameraController {

    @FXML private TableView<Camera> tableView;
    @FXML private TableColumn<Camera, Integer> idColumn;
    @FXML private TableColumn<Camera, String> statusColumn;
    @FXML private Label statusLabel;

    private final ObservableList<Camera> cameraList = FXCollections.observableArrayList();
    private final CameraService service = new CameraService();

    @FXML
    public void initialize() {
        idColumn.setCellValueFactory(new PropertyValueFactory<>("id"));
        statusColumn.setCellValueFactory(new PropertyValueFactory<>("status"));
        tableView.setItems(cameraList);
        refreshCameraList();
    }

    @FXML
    private void handleRefresh() {
        refreshCameraList();
    }

    @FXML
    private void handleStartSelected() {
        sendActionToSelected("START");
    }

    @FXML
    private void handleStopSelected() {
        sendActionToSelected("STOP");
    }

    @FXML
    private void handleStartAll() {
        sendActionToAll("START");
    }

    @FXML
    private void handleStopAll() {
        sendActionToAll("STOP");
    }

    private void refreshCameraList() {
        Task<List<Camera>> task = new Task<>() {
            @Override
            protected List<Camera> call() throws Exception {
                updateMessage("Fetching cameras...");
                return service.getAllCameras();
            }
        };
        task.setOnSucceeded(e -> {
            cameraList.setAll(task.getValue());
            statusLabel.setText("Camera list updated");
        });
        task.setOnFailed(e -> {
            Throwable ex = task.getException();
            statusLabel.setText("Error: " + ex.getMessage());
            showAlert("Error", "Failed to fetch cameras: " + ex.getMessage());
        });
        statusLabel.textProperty().bind(task.messageProperty());
        new Thread(task).start();
    }

    private void sendActionToSelected(String action) {
        Camera selected = tableView.getSelectionModel().getSelectedItem();
        if (selected == null) {
            showAlert("No Selection", "Please select a camera first.");
            return;
        }
        int id = selected.getId();
        Task<Void> task = new Task<>() {
            @Override
            protected Void call() throws Exception {
                updateMessage("Sending " + action + " to camera " + id + "...");
                service.sendCameraAction(id, action);
                return null;
            }
        };
        task.setOnSucceeded(e -> {
            statusLabel.setText("Command sent successfully");
            refreshCameraList();
        });
        task.setOnFailed(e -> {
            Throwable ex = task.getException();
            statusLabel.setText("Error: " + ex.getMessage());
            showAlert("Error", "Failed to send command: " + ex.getMessage());
        });
        statusLabel.textProperty().bind(task.messageProperty());
        new Thread(task).start();
    }

    private void sendActionToAll(String action) {
        Task<Void> task = new Task<>() {
            @Override
            protected Void call() throws Exception {
                updateMessage("Sending " + action + " to all cameras...");
                service.sendAllCamerasAction(action);
                return null;
            }
        };
        task.setOnSucceeded(e -> {
            statusLabel.setText("Command sent to all cameras");
            refreshCameraList();
        });
        task.setOnFailed(e -> {
            Throwable ex = task.getException();
            statusLabel.setText("Error: " + ex.getMessage());
            showAlert("Error", "Failed to send global command: " + ex.getMessage());
        });
        statusLabel.textProperty().bind(task.messageProperty());
        new Thread(task).start();
    }

    private void showAlert(String title, String content) {
        Platform.runLater(() -> {
            Alert alert = new Alert(Alert.AlertType.ERROR);
            alert.setTitle(title);
            alert.setHeaderText(null);
            alert.setContentText(content);
            alert.showAndWait();
        });
    }
}

package camera;

import java.io.*;
import java.net.HttpURLConnection;
import java.net.URL;
import java.nio.charset.StandardCharsets;
import java.util.ArrayList;
import java.util.List;

public class CameraService {
    private static final String BASE_URL = "http://localhost:8080/api";

    public List<Camera> getAllCameras() throws IOException {
        HttpURLConnection conn = createConnection(BASE_URL + "/cameras", "GET");
        if (conn.getResponseCode() != 200) throw new IOException("GET /cameras failed: " + conn.getResponseCode());
        String json = readResponse(conn);
        return parseCameraArray(json);
    }

    public Camera getCamera(int id) throws IOException {
        HttpURLConnection conn = createConnection(BASE_URL + "/cameras/" + id, "GET");
        if (conn.getResponseCode() != 200) throw new IOException("GET /cameras/" + id + " failed");
        String json = readResponse(conn);
        return parseSingleCamera(json);
    }

    public void sendCameraAction(int id, String action) throws IOException {
        String jsonBody = "{\"action\":\"" + action + "\"}";
        HttpURLConnection conn = createConnection(BASE_URL + "/cameras/" + id + "/record", "POST");
        conn.setRequestProperty("Content-Type", "application/json");
        conn.setDoOutput(true);
        try (OutputStream os = conn.getOutputStream()) {
            os.write(jsonBody.getBytes(StandardCharsets.UTF_8));
        }
        int code = conn.getResponseCode();
        if (code != 200 && code != 204) throw new IOException("POST action failed: " + code);
    }

    public void sendAllCamerasAction(String action) throws IOException {
        String jsonBody = "{\"action\":\"" + action + "\"}";
        HttpURLConnection conn = createConnection(BASE_URL + "/cameras/all/record", "POST");
        conn.setRequestProperty("Content-Type", "application/json");
        conn.setDoOutput(true);
        try (OutputStream os = conn.getOutputStream()) {
            os.write(jsonBody.getBytes(StandardCharsets.UTF_8));
        }
        int code = conn.getResponseCode();
        if (code != 200 && code != 204) throw new IOException("POST all action failed: " + code);
    }

    private HttpURLConnection createConnection(String urlString, String method) throws IOException {
        URL url = new URL(urlString);
        HttpURLConnection conn = (HttpURLConnection) url.openConnection();
        conn.setRequestMethod(method);
        conn.setConnectTimeout(5000);
        conn.setReadTimeout(5000);
        return conn;
    }

    private String readResponse(HttpURLConnection conn) throws IOException {
        try (BufferedReader br = new BufferedReader(
                new InputStreamReader(conn.getInputStream(), StandardCharsets.UTF_8))) {
            StringBuilder sb = new StringBuilder();
            String line;
            while ((line = br.readLine()) != null) sb.append(line);
            return sb.toString();
        }
    }

    private List<Camera> parseCameraArray(String json) {
        List<Camera> list = new ArrayList<>();
        json = json.trim();
        if (json.startsWith("[")) {
            json = json.substring(1, json.length() - 1).trim();
            if (json.isEmpty()) return list;
            String[] parts = json.split("\\},\\{");
            for (String part : parts) {
                part = part.replace("{", "").replace("}", "").trim();
                Camera c = parseCameraFromObject(part);
                if (c != null) list.add(c);
            }
        }
        return list;
    }

    private Camera parseSingleCamera(String json) {
        json = json.trim();
        if (json.startsWith("{") && json.endsWith("}")) {
            json = json.substring(1, json.length() - 1);
            return parseCameraFromObject(json);
        }
        return null;
    }

    private Camera parseCameraFromObject(String objStr) {
        String[] pairs = objStr.split(",");
        int id = -1;
        String status = null;
        for (String pair : pairs) {
            String[] kv = pair.split(":", 2);
            if (kv.length == 2) {
                String key = kv[0].trim().replace("\"", "");
                String value = kv[1].trim().replace("\"", "");
                if (key.equals("camerald")) id = Integer.parseInt(value);
                else if (key.equals("status")) status = value;
            }
        }
        return (id != -1 && status != null) ? new Camera(id, status) : null;
    }
}
